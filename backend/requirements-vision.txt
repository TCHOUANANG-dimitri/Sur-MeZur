# Dépendances de la chaîne de mesure par vision — OPTIONNELLES.
#
# Le backend démarre et fonctionne sans elles : la chaîne se désactive d'
# elle-même et l'estimation heuristique prend le relais. À installer seulement
# quand on veut activer la mesure réelle.
#
#   pip install -r requirements-vision.txt
#
# Vérifier ensuite l'état réel :  GET /api/measurements/capabilities

# --- MediaPipe : les 33 points du squelette -------------------------------
mediapipe==0.10.14
opencv-python-headless==4.10.0.84
# Pillow est requis pour la réduction d'image (pipeline._downscaled).
# Sans lui, les photos 3000-4000 px sont analysées en pleine résolution
# (87 s au lieu de 9 s — facteur 9,7x mesuré, voir pipeline.py:113-116).
Pillow>=10.0

# --- Modèle de mensurations (artefacts .joblib) ---------------------------
# scikit-learn DOIT correspondre à la version d'entraînement : un artefact
# rechargé sous une version différente peut échouer ou dériver silencieusement.
scikit-learn==1.7.2
joblib==1.5.3
numpy==2.2.6
# pandas permet de passer des colonnes nommées au modèle : sklearn valide alors
# les noms au lieu de se fier à l'ordre positionnel.
pandas>=2.0

# --- MobileSAM : silhouette (largeurs et profondeurs réelles) -------------
# Backend retenu en production (SAM_BACKEND=mobile_sam, voir config.py) :
# ~40 Mo de poids contre 375 Mo pour le SAM complet, et nettement plus rapide
# sur CPU (~1s de chargement, quelques secondes par image). Le SAM complet
# (segment-anything, torch avec CUDA) n'est plus installé ici — voir
# RAPPORT_PROJET.md pour le comparatif qui a motivé ce choix.
#
# --extra-index-url + suffixe +cpu : sans ça, pip installe la variante CUDA de
# torch par défaut (plusieurs Go de bibliothèques nvidia-*) alors qu'aucun GPU
# n'est disponible ici, ni en local ni sur Render.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.5.1+cpu
torchvision==0.20.1+cpu
timm
mobile-sam @ git+https://github.com/ChaoningZhang/MobileSAM.git
